    <script>
        pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/2.16.105/pdf.worker.min.js';

        let originalTextCache = '';
        let typedTextCache = '';

        function handleFileSelect(event, type) {
            const file = event.target.files[0];
            if (!file) return;

            if (type === 'orig') {
                document.getElementById('orig-file-name').innerText = file.name;
                extractTextFromPDF(file, (text) => { originalTextCache = text; });
            } else {
                document.getElementById('typed-file-name').innerText = file.name;
                extractTextFromPDF(file, (text) => { typedTextCache = text; });
            }
        }

        function extractTextFromPDF(file, callback) {
            const reader = new FileReader();
            reader.readAsArrayBuffer(file);
            reader.onload = function() {
                const typedarray = new Uint8Array(this.result);
                pdfjsLib.getDocument(typedarray).promise.then(async function(pdfDoc) {
                    let fullText = '';
                    for (let i = 1; i <= pdfDoc.numPages; i++) {
                        const page = await pdfDoc.getPage(i);
                        const textContent = await page.getTextContent();
                        const pageText = textContent.items.map(item => item.str).join(' ');
                        fullText += pageText + ' ';
                    }
                    callback(fullText.trim());
                });
            };
        }

        function processPDFsAndEvaluate() {
            if (!originalTextCache || !typedTextCache) {
                alert('Please upload both Original PDF and Transcribed PDF first!');
                return;
            }

            const origWords = originalTextCache.split(/\s+/);
            const typedWords = typedTextCache.split(/\s+/);

            let fullMistakes = 0;
            let halfMistakes = 0;
            let htmlOutput = '';
            let oIdx = 0, tIdx = 0;

            while (oIdx < origWords.length || tIdx < typedWords.length) {
                let oWord = origWords[oIdx] || '';
                let tWord = typedWords[tIdx] || '';

                if (oWord.toLowerCase() === tWord.toLowerCase()) {
                    htmlOutput += `<span class="word-correct">${tWord}</span> `;
                    oIdx++;
                    tIdx++;
                } else {
                    let nextOrig = origWords[oIdx + 1] || '';
                    let nextTyped = typedWords[tIdx + 1] || '';

                    if (oWord.toLowerCase() === nextTyped.toLowerCase()) {
                        htmlOutput += `<span class="word-added" title="Addition">+${tWord}</span> `;
                        fullMistakes++;
                        tIdx++;
                    } else if (nextOrig.toLowerCase() === tWord.toLowerCase()) {
                        htmlOutput += `<span class="word-missing" title="Omission">-${oWord}</span> `;
                        fullMistakes++;
                        oIdx++;
                    } else {
                        if (oWord && tWord) {
                            if (isSpellingError(oWord, tWord)) {
                                htmlOutput += `<span class="word-substituted" style="background:#fef3c7; color:#b45309;" title="Spelling Mistake">~${tWord}</span> `;
                                halfMistakes += 0.5;
                            } else {
                                htmlOutput += `<span class="word-substituted" title="Substitution">${tWord}</span> `;
                                fullMistakes++;
                            }
                        } else if (oWord) {
                            htmlOutput += `<span class="word-missing" title="Omission">-${oWord}</span> `;
                            fullMistakes++;
                        } else if (tWord) {
                            htmlOutput += `<span class="word-added" title="Addition">+${tWord}</span> `;
                            fullMistakes++;
                        }
                        oIdx++;
                        tIdx++;
                    }
                }
            }

            let totalWords = origWords.length;
            let totalEffectiveErrors = fullMistakes + halfMistakes;
            let errorPercent = ((totalEffectiveErrors / totalWords) * 100).toFixed(2);

            document.getElementById('total-words').innerText = totalWords;
            document.getElementById('full-mistakes').innerText = fullMistakes;
            document.getElementById('half-mistakes').innerText = halfMistakes;
            document.getElementById('effective-errors').innerText = totalEffectiveErrors;
            document.getElementById('error-percentage').innerText = errorPercent + '%';

            let statusEl = document.getElementById('ssc-status');
            if (errorPercent <= 5.0) {
                statusEl.innerText = 'PASS (Grade C & D)';
                statusEl.className = 'metric-value pass';
            } else if (errorPercent <= 7.0) {
                statusEl.innerText = 'PASS (Grade D Only)';
                statusEl.className = 'metric-value pass';
            } else {
                statusEl.innerText = 'FAIL (>7%)';
                statusEl.className = 'metric-value error-full';
            }

            document.getElementById('comparison-display').innerHTML = htmlOutput;
            document.getElementById('results-card').style.display = 'block';
            document.getElementById('results-card').scrollIntoView({ behavior: 'smooth' });
        }

        function isSpellingError(w1, w2) {
            if (Math.abs(w1.length - w2.length) > 2) return false;
            let diff = 0;
            for (let i = 0; i < Math.max(w1.length, w2.length); i++) {
                if (w1[i] !== w2[i]) diff++;
            }
            return diff <= 2;
        }
